## 二、数据采集脚本模板

### PowerShell版（行情/API采集）

```powershell
<#
.SYNOPSIS
    [脚本功能简述]
.DESCRIPTION
    [详细说明：数据源编号、API端点、返回格式]
.PARAMETER StockCodes
    股票代码列表
.PARAMETER OutputPath
    输出JSON路径
.EXAMPLE
    .\collect_xxx.ps1 -StockCodes @("000001","000002") -OutputPath "data.json"
#>
param(
    [Parameter(Mandatory=$true)]
    [string[]]$StockCodes,

    [Parameter(Mandatory=$false)]
    [string]$OutputPath = "output.json",

    [Parameter(Mandatory=$false)]
    [int]$DelayMs = 300
)

$ScriptName = "collect_xxx"
$LogPrefix = "[$ScriptName]"

# === 配置 ===
$ApiBaseUrl = "https://api.example.com"
$MaxRetries = 3
$BatchSize = 10

# === 函数 ===
function Invoke-ApiWithRetry {
    param([string]$Url, [int]$MaxRetries = 3)
    $retry = 0
    while ($retry -lt $MaxRetries) {
        try {
            $response = Invoke-RestMethod -Uri $Url -TimeoutSec 10
            return $response
        } catch {
            $retry++
            Write-Output "$LogPrefix 重试 $retry/$MaxRetries : $_"
            Start-Sleep -Milliseconds ($DelayMs * 2)
        }
    }
    Write-Output "$LogPrefix 请求失败(已重试$MaxRetries次): $Url"
    return $null
}

# === 主逻辑 ===
$results = @()
$total = $StockCodes.Count

for ($i = 0; $i -lt $total; $i++) {
    $code = $StockCodes[$i]
    Write-Output "$LogPrefix [$($i+1)/$total] 采集: $code"

    $data = Invoke-ApiWithRetry -Url "$ApiBaseUrl?code=$code"
    if ($data) {
        $results += $data
    }

    Start-Sleep -Milliseconds $DelayMs
}

# === 输出 ===
$results | ConvertTo-Json -Depth 10 | Out-File -FilePath $OutputPath -Encoding UTF8
Write-Output "$LogPrefix 完成: $($results.Count)/$total 条数据 → $OutputPath"
```

### Python版（复杂数据处理）

```python
"""
[脚本功能简述]
数据源：[编号] 来源名称
用法: python collect_xxx.py --codes 000001,000002 --output data.json
"""
import json
import time
import argparse
from typing import Optional
from datetime import datetime

# === 配置 ===
API_BASE_URL = "https://api.example.com"
MAX_RETRIES = 3
DELAY_SEC = 1.0
REQUEST_TIMEOUT = 10
SCRIPT_NAME = "collect_xxx"


def api_call_with_retry(url: str, max_retries: int = MAX_RETRIES) -> Optional[dict]:
    """API调用+重试[1]"""
    for attempt in range(max_retries):
        try:
            import requests
            resp = requests.get(url, timeout=REQUEST_TIMEOUT)
            resp.raise_for_status()
            return resp.json()
        except Exception as e:
            print(f"[{SCRIPT_NAME}] 重试 {attempt+1}/{max_retries}: {e}")
            time.sleep(DELAY_SEC * 2)
    print(f"[{SCRIPT_NAME}] 请求失败: {url}")
    return None


def main(codes: list[str], output_path: str):
    results = []
    total = len(codes)

    for i, code in enumerate(codes):
        print(f"[{SCRIPT_NAME}] [{i+1}/{total}] 采集: {code}")
        data = api_call_with_retry(f"{API_BASE_URL}?code={code}")
        if data:
            results.append(data)
        time.sleep(DELAY_SEC)

    with open(output_path, 'w', encoding='utf-8') as f:
        json.dump(results, f, ensure_ascii=False, indent=2)

    print(f"[{SCRIPT_NAME}] 完成: {len(results)}/{total} 条数据 → {output_path}")


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--codes", required=True, help="股票代码,逗号分隔")
    parser.add_argument("--output", default="output.json")
    args = parser.parse_args()
    main(args.codes.split(","), args.output)
```

## 五、批量处理工具模板

<#
.SYNOPSIS
    批量[操作描述]——遍历N只股票执行[操作]
#>
param(
    [string]$StockListPath = "stock_list.json",
    [string]$LogPath = "batch.log"
)

$stocks = Get-Content $StockListPath | ConvertFrom-Json
$total = $stocks.Count
$success = 0
$failed = 0

foreach ($i in 0..($total-1)) {
    $stock = $stocks[$i]
    try {
        # TODO: 具体操作
        Write-Output "[$($i+1)/$total] $stock : OK"
        $success++
    } catch {
        Write-Output "[$($i+1)/$total] $stock : FAILED - $_"
        $failed++
        Add-Content -Path $LogPath -Value "$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss') $stock : $_"
    }
    Start-Sleep -Milliseconds 300
}

Write-Output "完成: 成功$success 失败$failed"
